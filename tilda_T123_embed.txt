    loading="eager"
    allow="clipboard-write"
  ></iframe>

<script>
(function(){
  var frame = document.getElementById('campbox-sales-board-frame');
  if(!frame) return;
  window.addEventListener('message', function(event){
    if(event.source !== frame.contentWindow) return;
    var data = event.data || {};
    if(data.type !== 'campbox-salesboard-height') return;
    var h = Number(data.height || 0);
    if(!Number.isFinite(h)) return;
    h = Math.max(720, Math.min(h, 12000));
    frame.style.height = Math.ceil(h) + 'px';
  });
})();
</script>
